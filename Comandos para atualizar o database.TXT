                // Aguarda o carregamento completo do DOM
                document.addEventListener('DOMContentLoaded', function() {
                    // Seleciona todas as opções de resposta
                    const options = document.querySelectorAll('.list-group-item');
                    // Seleciona o elemento onde a mensagem de feedback será exibida
                    const feedbackMessage = document.getElementById('feedbackMessage');
        
                    // Adiciona um ouvinte de evento de clique a cada opção
                    options.forEach(option => {
                        option.addEventListener('click', function() {
                            // Verifica se a opção já foi clicada (para evitar múltiplos cliques)
                            if (this.classList.contains('correct') || this.classList.contains('incorrect')) {
                                return; // Sai da função se já houver uma classe de feedback
                            }
        
                            // Obtém o valor do atributo data-correct
                            const isCorrect = this.dataset.correct === 'true';
        
                            // Remove o cursor de ponteiro de todas as opções
                            options.forEach(opt => {
                                opt.style.cursor = 'default';
                            });
        
                            // Aplica o estilo e exibe a mensagem de feedback
                            if (isCorrect) {
                                this.classList.add('correct'); // Adiciona classe para estilo de correto
                                feedbackMessage.textContent = "Correta! Parabéns!";
                                feedbackMessage.classList.add('correct');
                            } else {
                                this.classList.add('incorrect'); // Adiciona classe para estilo de incorreto
                                feedbackMessage.textContent = "Incorreta. Tente novamente.";
                                feedbackMessage.classList.add('incorrect');
                                // Opcional: realçar a resposta correta após uma tentativa incorreta
                                document.querySelector('.list-group-item[data-correct="true"]').classList.add('correct');
                            }
                            feedbackMessage.style.display = 'block'; // Torna a mensagem visível
        
                            // Desabilita cliques em todas as opções após a primeira seleção
                            options.forEach(opt => {
                                opt.style.pointerEvents = 'none';
                            });
                        });
                    });
                });
            </script>
